create database ecommerce_db;
use ecommerce_db;
create table products (
    product_id   int primary key,
    product_name varchar(100),
    category     varchar(50),
    brand        varchar(50),
    price        decimal(10,2),
    quantity     int,
    city         varchar(50),
    status       varchar(20)
);
